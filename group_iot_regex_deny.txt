# Abused / throw-away TLDs that IoT devices have no reason to resolve
(\.|^)(zip|mov|top|xyz|cyou|icu|rest|quest|bond|sbs|cfd|click|buzz|monster|gq|tk|ml|cf|ga)$

# Free reverse-proxy / tunnel services often used for C2 and data exfiltration
(\.|^)(ngrok(-free)?\.(io|app|dev)|trycloudflare\.com|serveo\.net|localtunnel\.me|loca\.lt|pagekite\.me)$

# Paste / file-drop sites used to stage malware
(\.|^)(pastebin\.com|paste\.ee|hastebin\.com|transfer\.sh|anonfiles\.com|file\.io)$

# Raw IP-literal style names (e.g. 1-2-3-4.nip.io, sslip.io wildcards)
(\.|^)(nip\.io|sslip\.io|xip\.io)$
